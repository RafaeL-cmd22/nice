.data 
prompt1: .asciiz "Digite o primeiro número inteiro: "
prompt2: .asciiz "Digite o segundo número inteiro: "
result_msg: .asciiz "A diferença do maior para o menor é: "

.text
.globl main

main: 


    li $v0, 4
    la $a0, prompt1
    syscall
    li $v0, 5
    syscall
    move $t0, $v0        


    li $v0, 4
    la $a0, prompt2
    syscall
    li $v0, 5
    syscall
    move $t1, $v0          



    bge $t0, $t1, sub_t0_t1


    sub $t2, $t1, $t0      
    j print_result        

sub_t0_t1:
    sub $t2, $t0, $t1     

print_result:
    
    li $v0, 4
    la $a0, result_msg
    syscall
    
    
    li $v0, 1
    move $a0, $t2         
    syscall

    
    li $v0, 10
    syscall
